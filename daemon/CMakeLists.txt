cmake_minimum_required(VERSION 3.16)
project(telemetryd LANGUAGES CXX)
if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "telemetryd currently supports macOS and Linux.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(telemetryd src/main.cpp src/collector.cpp src/access.cpp)
target_compile_options(telemetryd PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(telemetryd PRIVATE Threads::Threads)
if(APPLE)
  target_link_libraries(telemetryd PRIVATE "-framework IOKit" "-framework CoreFoundation")
endif()
